% a vector cell: #1 name, #2 x, #3 row, #4 text, #5 highlighted?(1/0)

  \ifnum#5=1
    \node[vec, draw=tumdark, line width=1.8pt, fill=\c!75!white] at (#2,{\RY{#3}}) {#4};

% ===== group bands =====
\foreach \g/\ra/\rb in {0/0/2, 1/3/4}{

  \draw[\bc, dashed, line width=.7pt, rounded corners=4pt, fill=\bp!65!white]
    (0.05,{\RY{\ra}+0.45-0.03}) rectangle (\Wd,{\RY{\rb}-0.45+0.03});
  \node[\bc, font=\bfseries\footnotesize] at (0.75,{(\RY{\ra}+\RY{\rb})/2}) {Group \bn};

% ===== (1) input patch embeddings =====
\node[head] at (\Gx+2,-0.75) {Patch embeddings $\mathbf{H}$ (columns = patch index)};
\node[lab, gry] at (\Gx+2,-1.2) {$B$ series $\times$ $N$ patches};
\foreach \r in {0,...,4}{
  \node[lab] at (2.05,{\RY{\r}}) {series \fpeval{\r+1}};
  \foreach \cx/\on in {0.5/0, 1.5/0, 3.5/1}{
    \edef\c{\gc{\r}}
    \ifnum\on=1
      \node[vec, draw=\c, fill=\c, minimum width=.82cm] at (\Gx+\cx,{\RY{\r}}) {};
    \else
      \node[vec, draw=\c, fill=\c!35!white, minimum width=.82cm] at (\Gx+\cx,{\RY{\r}}) {};
    \fi
  }
  \node[lab, gry] at (\Gx+2.5,{\RY{\r}}) {$\cdots$};
}
\node[lab, gry] at (\Gx+0.5,{-\Yzero+0.62}) {1};
\node[lab, gry] at (\Gx+1.5,{-\Yzero+0.62}) {2};

  (\Gx+3.5-0.55,{\RY{0}+0.45+0.02}) rectangle (\Gx+3.5+0.55,{\RY{4}-0.45-0.02});
